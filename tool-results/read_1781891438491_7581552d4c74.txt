   844→            detectText.innerHTML = 'Great! Your browser supports app installation. <strong>Click "Install Now"</strong> below or use the banner at the bottom.';
   845→            detectBanner.style.display = 'block';
   846→        }
   847→    });

   849→    function installPWA() {
   850→        if (deferredPrompt) {
   851→            deferredPrompt.prompt();
   852→            deferredPrompt.userChoice.then((choiceResult) => {
   853→                if (choiceResult.outcome === 'accepted') {
   854→                    console.log('[PWA] User accepted the install prompt');
   855→                    dismissBanner();
   856→                } else {
   857→                    console.log('[PWA] User dismissed the install prompt');
   858→                }
   859→                deferredPrompt = null;
   860→            });
   861→        } else {
   862→            // Fallback: direct to login page so they can install from there

   864→        }
   865→    }
