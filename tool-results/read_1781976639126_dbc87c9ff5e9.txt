    20→    public function index(Request $request)
    21→    {
    22→        $user = Auth::user();
    23→        $teacher = $user->teacherProfile ?? \App\Models\Teacher::where('email', $user->email)->first();
    24→
    25→        $query = AssessmentQuestion::with(['subject', 'classroom', 'section', 'options', 'answers']);
    26→
    27→        // Non-admin: only see own questions
    28→        if (!in_array($user->role, ['admin', 'super_admin', 'branch_principal', 'general_manager'])) {
    29→            if ($teacher) {
    30→                $query->where('teacher_id', $teacher->id);
    31→            }
    32→        }
    33→
    34→        // Filters
    35→        if ($request->filled('subject_id')) {
    36→            $query->where('subject_id', $request->subject_id);
    37→        }
    38→        if ($request->filled('class_id')) {
    39→            $query->where('class_id', $request->class_id);
    40→        }
    41→        if ($request->filled('difficulty')) {
    42→            $query->where('difficulty', $request->difficulty);
    43→        }
    44→        if ($request->filled('question_type')) {
    45→            $query->where('question_type', $request->question_type);
    46→        }
    47→        if ($request->filled('topic')) {
    48→            $query->where('topic', 'like', "%{$request->topic}%");
    49→        }
    50→        if ($request->filled('search')) {
    51→            $search = $request->search;
    52→            $query->where(function ($q) use ($search) {
    53→                $q->where('question_text', 'like', "%{$search}%")
    54→                  ->orWhere('title', 'like', "%{$search}%")
    55→                  ->orWhere('topic', 'like', "%{$search}%");
    56→            });
    57→        }
    58→
    59→        $questions = $query->latest()->paginate(20);
    60→
    61→        // Get subjects the teacher is assigned to
    62→        $activeAy = AcademicYear::where('is_current', true)->first();
    63→        $subjects = $this->getTeacherSubjects($teacher, $activeAy);
    64→        $classes = $this->getTeacherClasses($teacher, $activeAy);
    65→        $topics = AssessmentQuestion::when($teacher, fn($q) => $q->where('teacher_id', $teacher->id))
    66→            ->whereNotNull('topic')->distinct()->orderBy('topic')->pluck('topic');
    67→
    68→        return view('admin.assessment-questions.index', compact(
    69→            'questions', 'subjects', 'classes', 'topics', 'teacher', 'activeAy'
    70→        ));
    71→    }

    89→    public function store(Request $request)
    90→    {
    91→        $user = Auth::user();
    92→        $teacher = $user->teacherProfile ?? \App\Models\Teacher::where('email', $user->email)->first();
    93→
    94→        if (!$teacher && !in_array($user->role, ['admin', 'super_admin'])) {

    97→
    98→        $validated = $request->validate([
    99→            'title' => 'nullable|string|max:255',
   100→            'question_text' => 'required|string|max:5000',
   101→            'question_type' => 'required|in:multiple_choice,true_false,short_answer',
   102→            'subject_id' => 'required|exists:subjects,id',
   103→            'class_id' => 'required|exists:classes,id',
   104→            'difficulty' => 'required|in:easy,medium,hard',
   105→            'topic' => 'nullable|string|max:255',
   106→            'marks' => 'integer|min:1|max:100',
   107→            'hint' => 'nullable|string|max:2000',
   108→            'explanation' => 'nullable|string|max:10000',
   109→            'worked_out_solution' => 'nullable|string|max:20000',
   110→            'is_active' => 'boolean',
   111→            // Options for multiple choice
   112→            'options' => 'required_if:question_type,multiple_choice|array|min:2|max:6',
   113→            'options.*.option_text' => 'required_with:options|string|max:1000',
   114→            'options.*.option_label' => 'required_with:options|string|max:1',
   115→            'options.*.is_correct' => 'boolean',
   116→            // True/False
   117→            'correct_tf' => 'required_if:question_type,true_false|in:true,false',
   118→            // Safe Exam Browser
   119→            'seb_mode' => 'nullable|in:none,optional,required',
   120→            'seb_browser_view_mode' => 'nullable|integer|in:0,1,2',
   121→            'seb_allow_quit' => 'nullable|boolean',
   122→            'seb_quit_password' => 'nullable|string|max:255',
   123→            'seb_show_taskbar' => 'nullable|boolean',
   124→            'seb_show_time' => 'nullable|boolean',
   125→            'seb_allow_spell_check' => 'nullable|boolean',
   126→            'seb_allowed_urls_text' => 'nullable|string|max:5000',
   127→        ]);
   128→
   129→        $activeAy = AcademicYear::where('is_current', true)->first();
   130→
   131→        // Parse SEB allowed URLs from textarea
   132→        $sebAllowedUrls = null;
   133→        if ($request->filled('seb_allowed_urls_text')) {
   134→            $urls = array_filter(array_map('trim', explode("\n", $request->seb_allowed_urls_text)));
   135→            $sebAllowedUrls = !empty($urls) ? $urls : null;
   136→        }
   137→
   138→        // Generate SEB config key if SEB mode is enabled
   139→        $sebConfigKey = null;
   140→        $sebMode = $request->input('seb_mode', 'none');
   141→        if (in_array($sebMode, ['optional', 'required'])) {
   142→            $sebConfigKey = bin2hex(random_bytes(32));
   143→        }
   144→
   145→        // Questions are class-level — apply to ALL branches and ALL sections
   146→        $question = AssessmentQuestion::create([
   147→            'teacher_id' => $teacher?->id,
   148→            'subject_id' => $validated['subject_id'],
   149→            'class_id' => $validated['class_id'],
   150→            'section_id' => null, // null = applies to all sections
   151→            'academic_year_id' => $activeAy?->id,
   152→            'branch_id' => null, // null = applies to all branches
   153→            'title' => $validated['title'] ?? null,
   154→            'question_text' => $validated['question_text'],
   155→            'question_type' => $validated['question_type'],
   156→            'hint' => $validated['hint'] ?? null,
   157→            'explanation' => $validated['explanation'] ?? null,
   158→            'worked_out_solution' => $validated['worked_out_solution'] ?? null,
   159→            'difficulty' => $validated['difficulty'],
   160→            'topic' => $validated['topic'] ?? null,
   161→            'marks' => $validated['marks'] ?? 1,
   162→            'is_active' => $request->boolean('is_active', true),
   163→            // SEB fields
   164→            'seb_mode' => $sebMode,
   165→            'seb_config_key' => $sebConfigKey,
   166→            'seb_browser_view_mode' => $request->input('seb_browser_view_mode', 1),
   167→            'seb_allow_quit' => $request->boolean('seb_allow_quit', false),
   168→            'seb_quit_password' => $request->input('seb_quit_password'),
   169→            'seb_show_taskbar' => $request->boolean('seb_show_taskbar', true),
   170→            'seb_show_time' => $request->boolean('seb_show_time', true),
   171→            'seb_allow_spell_check' => $request->boolean('seb_allow_spell_check', false),
   172→            'seb_allowed_urls' => $sebAllowedUrls,
   173→        ]);
   174→
   175→        // Create options based on question type
   176→        if ($validated['question_type'] === 'multiple_choice' && !empty($validated['options'])) {
   177→            foreach ($validated['options'] as $idx => $opt) {
   178→                AssessmentOption::create([
   179→                    'assessment_question_id' => $question->id,
   180→                    'option_text' => $opt['option_text'],
   181→                    'option_label' => $opt['option_label'] ?? chr(65 + $idx), // A, B, C...
   182→                    'is_correct' => isset($opt['is_correct']) && $opt['is_correct'],
   183→                    'sort_order' => $idx,
   184→                ]);
   185→            }
   186→        } elseif ($validated['question_type'] === 'true_false') {
   187→            AssessmentOption::create([
   188→                'assessment_question_id' => $question->id,
   189→                'option_text' => 'True',
   190→                'option_label' => 'A',
   191→                'is_correct' => $validated['correct_tf'] === 'true',
   192→                'sort_order' => 0,
   193→            ]);
   194→            AssessmentOption::create([
   195→                'assessment_question_id' => $question->id,
   196→                'option_text' => 'False',
   197→                'option_label' => 'B',
   198→                'is_correct' => $validated['correct_tf'] === 'false',
   199→                'sort_order' => 1,
   200→            ]);
   201→        }
   202→
   203→        return redirect()->route('admin.assessment-questions.index')
   204→            ->with('success', 'Question created successfully.');
   205→    }

   567→    // ── Import questions from CSV upload ─────────────────────

   569→    public function bulkImport(Request $request)
   570→    {
   571→        $request->validate([
   572→            'import_file' => 'required|file|mimes:csv,txt,xlsx,xls|max:5120',
   573→        ]);
   574→
   575→        $user = Auth::user();
   576→        $teacher = $user->teacherProfile ?? \App\Models\Teacher::where('email', $user->email)->first();
   577→
   578→        if (!$teacher && !in_array($user->role, ['admin', 'super_admin'])) {
   579→            return back()->with('error', 'Teacher profile not found.')->withInput();
   580→        }
   581→
   582→        $activeAy = AcademicYear::where('is_current', true)->first();
   583→        $file = $request->file('import_file');
   584→        $path = $file->getRealPath();
   585→
   586→        $imported = 0;
   587→        $skipped = 0;
   588→        $errors = [];
   589→
   590→        try {
   591→            // Handle CSV files
   592→            if ($file->getClientOriginalExtension() === 'csv' || $file->getClientOriginalExtension() === 'txt') {
   593→                $handle = fopen($path, 'r');
   594→                $header = fgetcsv($handle); // Skip header row
   595→
   596→                $rowNum = 1;
   597→                while (($row = fgetcsv($handle)) !== false) {
   598→                    $rowNum++;
   599→                    try {
   600→                        $result = $this->importQuestionRow($row, $teacher, $activeAy, $user, $rowNum);
   601→                        if ($result === true) {
   602→                            $imported++;
   603→                        } else {
   604→                            $skipped++;
   605→                            $errors[] = $result;
   606→                        }
   607→                    } catch (\Exception $e) {
   608→                        $skipped++;
   609→                        $errors[] = "Row {$rowNum}: " . $e->getMessage();
   610→                    }
   611→                }
   612→                fclose($handle);
   613→            } else {
   614→                // Handle Excel files using PhpSpreadsheet
   615→                if (!class_exists('\PhpOffice\PhpSpreadsheet\IOFactory')) {
   616→                    return back()->with('error', 'Excel file support requires PhpSpreadsheet. Please upload a CSV file instead.')->withInput();
   617→                }
   618→                $spreadsheet = \PhpOffice\PhpSpreadsheet\IOFactory::load($path);
   619→                $worksheet = $spreadsheet->getActiveSheet();
   620→                $rows = $worksheet->toArray();
   621→
   622→                // Skip header row
   623→                for ($i = 1; $i < count($rows); $i++) {
   624→                    $rowNum = $i + 1;
   625→                    $row = $rows[$i];
   626→                    try {
   627→                        $result = $this->importQuestionRow($row, $teacher, $activeAy, $user, $rowNum);
   628→                        if ($result === true) {
   629→                            $imported++;
   630→                        } else {
   631→                            $skipped++;
   632→                            $errors[] = $result;
   633→                        }
   634→                    } catch (\Exception $e) {
   635→                        $skipped++;
   636→                        $errors[] = "Row {$rowNum}: " . $e->getMessage();
   637→                    }
   638→                }
   639→            }
   640→        } catch (\Exception $e) {
   641→            return back()->with('error', 'Failed to read file: ' . $e->getMessage())->withInput();
   642→        }
   643→
   644→        $message = "{$imported} questions imported successfully.";
   645→        if ($skipped > 0) {
   646→            $message .= " {$skipped} rows skipped.";
   647→        }
   648→
   649→        return redirect()->route('admin.assessment-questions.index')
   650→            ->with('success', $message)
   651→            ->with('import_errors', $errors);
   652→    }

   654→    /**
   655→     * Import a single question row from CSV/Excel.
   656→     * Returns true on success, or an error message string on failure.
   657→     */
   658→    private function importQuestionRow($row, $teacher, $activeAy, $user, $rowNum)
   659→    {
   660→        $questionText = trim($row[0] ?? '');
   661→        $questionType = trim($row[1] ?? 'multiple_choice');
   662→        $subjectName  = trim($row[2] ?? '');
   663→        $className    = trim($row[3] ?? '');
   664→        // Column 4 (section_name) is no longer used — questions are class-level
   665→        $difficulty   = trim($row[4] ?? 'medium');
   666→        if (empty($difficulty) || !in_array($difficulty, ['easy', 'medium', 'hard'])) {
   667→            // If column 4 looks like a difficulty value, it was an old-format row
   668→            // where section_name was column 4 and difficulty was column 5
   669→            // Try reading from column 5 instead
   670→            $possibleDifficulty = trim($row[5] ?? '');
   671→            if (in_array($possibleDifficulty, ['easy', 'medium', 'hard'])) {
   672→                $difficulty = $possibleDifficulty;
   673→                $colOffset = 1; // old format has extra column
   674→            } else {
   675→                $difficulty = 'medium';
   676→                $colOffset = 0;
   677→            }
   678→        } else {
   679→            $colOffset = 0; // new format (no section_name column)
   680→        }
   681→        $marks        = intval($row[5 + $colOffset] ?? 1);
   682→        $topic        = trim($row[6 + $colOffset] ?? '');
   683→        $title        = trim($row[7 + $colOffset] ?? '');
   684→        $hint         = trim($row[8 + $colOffset] ?? '');
   685→        $optionA      = trim($row[9 + $colOffset] ?? '');
   686→        $optionB      = trim($row[10 + $colOffset] ?? '');
   687→        $optionC      = trim($row[11 + $colOffset] ?? '');
   688→        $optionD      = trim($row[12 + $colOffset] ?? '');
   689→        $correctOption = strtoupper(trim($row[13 + $colOffset] ?? ''));
   690→        $explanation  = trim($row[14 + $colOffset] ?? '');
   691→        $workedOutSolution = trim($row[15 + $colOffset] ?? '');
   692→
   693→        // Validate required fields
   694→        if (empty($questionText)) {
   695→            return "Row {$rowNum}: Question text is empty.";
   696→        }
   697→
   698→        if (!in_array($questionType, ['multiple_choice', 'true_false', 'short_answer'])) {
   699→            return "Row {$rowNum}: Invalid question type '{$questionType}'. Use: multiple_choice, true_false, or short_answer.";
   700→        }
   701→
   702→        if (!in_array($difficulty, ['easy', 'medium', 'hard'])) {
   703→            $difficulty = 'medium';
   704→        }
   705→
   706→        // Resolve subject
   707→        $subject = null;
   708→        if (!empty($subjectName)) {
   709→            $subject = Subject::where('name', $subjectName)->first();
   710→            if (!$subject) {
   711→                return "Row {$rowNum}: Subject '{$subjectName}' not found.";
   712→            }
   713→        } else {
   714→            // Try first subject from teacher's assignments
   715→            $subjects = $this->getTeacherSubjects($teacher, $activeAy);
   716→            $subject = $subjects->first();
   717→            if (!$subject) {
   718→                return "Row {$rowNum}: No subject specified and no teacher assignments found.";
   719→            }
   720→        }
   721→
   722→        // Resolve class
   723→        $class = null;
   724→        if (!empty($className)) {
   725→            $class = ClassRoom::where('name', $className)->first();
   726→            if (!$class) {
   727→                return "Row {$rowNum}: Class '{$className}' not found.";
   728→            }
   729→        } else {
   730→            $classes = $this->getTeacherClasses($teacher, $activeAy);
   731→            $class = $classes->first();
   732→            if (!$class) {
   733→                return "Row {$rowNum}: No class specified and no teacher assignments found.";
   734→            }
   735→        }
   736→
   737→        // Questions are class-level — apply to ALL branches and ALL sections
   738→        // Create the question
   739→        $question = AssessmentQuestion::create([
   740→            'teacher_id' => $teacher?->id,
   741→            'subject_id' => $subject->id,
   742→            'class_id' => $class->id,
   743→            'section_id' => null, // null = applies to all sections
   744→            'academic_year_id' => $activeAy?->id,
   745→            'branch_id' => null, // null = applies to all branches
   746→            'title' => $title ?: null,
   747→            'question_text' => $questionText,
   748→            'question_type' => $questionType,
   749→            'hint' => $hint ?: null,
   750→            'explanation' => $explanation ?: null,
   751→            'worked_out_solution' => $workedOutSolution ?: null,
   752→            'difficulty' => $difficulty,
   753→            'topic' => $topic ?: null,
   754→            'marks' => max(1, min(100, $marks)),
   755→            'is_active' => true,
   756→        ]);
   757→
   758→        // Create options based on question type
   759→        if ($questionType === 'multiple_choice') {
   760→            $options = [];
   761→            if (!empty($optionA)) $options[] = ['text' => $optionA, 'label' => 'A'];
   762→            if (!empty($optionB)) $options[] = ['text' => $optionB, 'label' => 'B'];
   763→            if (!empty($optionC)) $options[] = ['text' => $optionC, 'label' => 'C'];
   764→            if (!empty($optionD)) $options[] = ['text' => $optionD, 'label' => 'D'];
   765→
   766→            if (count($options) < 2) {
   767→                $question->delete();
   768→                return "Row {$rowNum}: Multiple choice needs at least 2 options (A and B).";
   769→            }
   770→
   771→            foreach ($options as $idx => $opt) {
   772→                AssessmentOption::create([
   773→                    'assessment_question_id' => $question->id,
   774→                    'option_text' => $opt['text'],
   775→                    'option_label' => $opt['label'],
   776→                    'is_correct' => $opt['label'] === $correctOption,
   777→                    'sort_order' => $idx,
   778→                ]);
   779→            }
   780→        } elseif ($questionType === 'true_false') {
   781→            $isTrue = in_array(strtolower($correctOption), ['true', 'a', 'yes', '1']);
   782→            AssessmentOption::create([
   783→                'assessment_question_id' => $question->id,
   784→                'option_text' => 'True',
   785→                'option_label' => 'A',
   786→                'is_correct' => $isTrue,
   787→                'sort_order' => 0,
   788→            ]);
   789→            AssessmentOption::create([
   790→                'assessment_question_id' => $question->id,
   791→                'option_text' => 'False',
   792→                'option_label' => 'B',
   793→                'is_correct' => !$isTrue,
   794→                'sort_order' => 1,
   795→            ]);
   796→        }
   797→
   798→        return true;
   799→    }

   841→    // ── Helpers ─────────────────────────────────────────────
   842→
   843→    private function getTeacherSubjects($teacher, $activeAy)
   844→    {
   845→        if (!$teacher || !$activeAy) {
   846→            return Subject::orderBy('name')->get();
   847→        }
   848→
   849→        // Primary lookup: teacher_id references teachers.id
   850→        $subjectIds = TeacherAssignment::where('teacher_id', $teacher->id)
   851→            ->where('academic_year_id', $activeAy->id)
   852→            ->pluck('subject_id')
   853→            ->unique();
   854→
   855→        // Fallback: if no assignments found via teacher->id, try via user_id
   856→        if ($subjectIds->isEmpty() && $teacher->user_id) {
   857→            $subjectIds = TeacherAssignment::where('teacher_id', $teacher->user_id)
   858→                ->where('academic_year_id', $activeAy->id)
   859→                ->pluck('subject_id')
   860→                ->unique();
   861→        }
   862→
   863→        // Final fallback: try matching by email through users table
   864→        if ($subjectIds->isEmpty() && $teacher->email) {
   865→            $user = \App\Models\User::where('email', $teacher->email)->first();
   866→            if ($user) {
   867→                $subjectIds = TeacherAssignment::where('teacher_id', $user->id)
   868→                    ->where('academic_year_id', $activeAy->id)
   869→                    ->pluck('subject_id')
   870→                    ->unique();
   871→            }
   872→        }
   873→
   874→        return Subject::whereIn('id', $subjectIds)->orderBy('name')->get();
   875→    }

   877→    private function getTeacherClasses($teacher, $activeAy)
   878→    {

   885→            ->where('academic_year_id', $activeAy->id)
   886→            ->pluck('class_id')

   890→        // This handles the case where teacher_assignments.teacher_id still
   891→        // stores users.id values (before the data-fix migration runs).
   892→        if ($classIds->isEmpty() && $teacher->user_id) {
   893→            $classIds = TeacherAssignment::where('teacher_id', $teacher->user_id)
   894→                ->where('academic_year_id', $activeAy->id)

   899→        // Final fallback: if still empty, try matching by email through users table
   900→        if ($classIds->isEmpty() && $teacher->email) {

   902→            if ($user) {
