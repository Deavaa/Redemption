   665→function exportCSV(){
   666→    var tables=document.querySelectorAll('.fms-seq-table');
   667→    if(!tables.length)return;
   668→    var csv=[];
   669→    tables.forEach(function(table){
   670→        var sectionHead=table.closest('.fms-term-section');
   671→        if(sectionHead){
   672→            var headDiv=sectionHead.querySelector('.fms-term-head');
   673→            if(headDiv){
   674→                csv.push('');
   675→                csv.push('=== '+headDiv.innerText.trim()+' ===');
   676→                csv.push('');
   677→            }
   678→        }
   679→        var rows=table.querySelectorAll('tr');
   680→        rows.forEach(function(row){
   681→            var cols=row.querySelectorAll('td,th');
   682→            var rowData=[];
   683→            cols.forEach(function(col){
   684→                var text=col.innerText.replace(/"/g,'""').replace(/\n/g,' ');
   685→                rowData.push('"'+text+'"');
   686→            });
   687→            csv.push(rowData.join(','));
   688→        });
   689→    });
   690→    var blob=new Blob([csv.join('\n')],{type:'text/csv;charset=utf-8;'});
   691→    var link=document.createElement('a');
   692→    link.href=URL.createObjectURL(blob);
   693→    link.download='mark_sheet_full.csv';
   694→    link.click();
   695→}
