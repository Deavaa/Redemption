    24→     * Resolve the logged-in user's Teacher record.
    25→     * Tries user_id FK first, then falls back to email match.
    26→     * Returns null for non-teacher users or if no Teacher record found.
    27→     */
    28→    private function getTeacherForUser()
    29→    {
    30→        $user = auth()->user();
    31→        if (!$user || $user->role !== 'teacher') return null;
    32→
    33→        // Try user_id FK first
    34→        $teacher = Teacher::where('user_id', $user->id)->first();
    35→        if (!$teacher) {
    36→            // Fall back to email match (legacy)
    37→            $teacher = Teacher::where('email', $user->email)->first();
    38→        }
    39→        return $teacher;
    40→    }

   203→                ?? AcademicYear::orderBy('id', 'desc')->first();
   204→            $ayId = $currentAy?->id;

   208→            // Only return classes from teacher's assignments + homeroom classes
   209→            $assignmentClassIds = $teacher->assignments()->pluck('class_id')->unique();
   210→            $homeroomClassIds = $teacher->classRooms()->pluck('id');
   211→            $classIds = $assignmentClassIds->merge($homeroomClassIds)->unique();

   213→            $query = ClassRoom::whereIn('id', $classIds);
   214→            if ($ayId) {

   236→            $query->where('academic_year_id', $ayId);
   237→        }
   238→        if ($effectiveBranchId) {
   239→            $query->where('branch_id', $effectiveBranchId);

   292→                // Only return sections from teacher's assignments for this class
   293→                // + sections where they are homeroom teacher
   294→                $assignmentSectionIds = $teacher->assignments()
   295→                    ->where('class_id', $classId)
   296→                    ->pluck('section_id')
   297→                    ->filter()
   298→                    ->unique();
   299→                $homeroomSectionIds = $teacher->sections()
   300→                    ->where('class_id', $classId)
   301→                    ->pluck('id');
   302→
   303→                $allowedSectionIds = $assignmentSectionIds->merge($homeroomSectionIds)->unique();
   304→
   305→                // If teacher has assignments with null section_id for this class, they can see all sections
   306→                $hasNullSectionAssignment = $teacher->assignments()
   307→                    ->where('class_id', $classId)
   308→                    ->whereNull('section_id')
   309→                    ->exists();
   310→
   311→                if (!$hasNullSectionAssignment) {
   312→                    $query->whereIn('id', $allowedSectionIds);
   313→                }
   314→                // If they have null-section assignment, they see all sections in that class
   315→            }
   316→
   317→            return response()->json($query->orderBy('name','asc')->get(['id','name']));
   318→        }
   319→
   320→        if ($classGrade) {
   321→            $sections = Student::where('class_grade',$classGrade)
   322→                ->whereNotNull('section')
   323→                ->where('section','<>','')
   324→                ->distinct()
   325→                ->orderBy('section','asc')
   326→                ->pluck('section');
   327→            return response()->json($sections);
   328→        }
   329→
   330→        return response()->json([]);
   331→    }

   335→        $sectionId = $request->query('section_id');

   350→
   351→        // Filter by academic year: show AY-specific + no-AY assignments

   362→            $isHomeroomSection = $sectionId ? $teacher->sections()->where('id', $sectionId)->exists() : false;

   365→                // Not homeroom — only show their own assigned subjects

   369→        }
   370→
   371→        $assignments = $query->get();

   373→            $subj = $a->subject;

   454→            }

   501→        $caWeight = MarkEntryConfig::getCaWeight();
   502→        $examWeight = MarkEntryConfig::getExamWeight();
   503→        $precision = MarkEntryConfig::getRoundingPrecision();
   504→        // Calculate CA raw total from field max values
   505→        $caRawTotal = 0;
   506→        foreach ($markFields as $field) {
   507→            $cat = $field['category'] ?? 'ca';
   508→            if ($cat === 'ca' || $cat === 'extra_ca') {
   509→                $caRawTotal += $field['max'];
   510→            }
   511→        }
   512→        $rows = [];
   513→        foreach ($students as $s) {
   514→            $mark = $existingMarks->get($s->student_id);
   515→            $row = ['student_id'=>$s->student_id,'student_name'=>$s->student_name??'Unknown','roll_number'=>$s->roll_number,'gender'=>$s->gender];
   516→            foreach ($markFields as $field) { $col=$field['col']; $row[$col]=$mark?floatval($mark->$col):null; }
   517→            // Recalculate totals from raw fields (don't trust stored values)
   518→            if ($mark) {
   519→                $caRaw = 0;
   520→                foreach ($caFields as $f) { $caRaw += floatval($mark->$f ?? 0); }
   521→                $examRaw = 0;
   522→                foreach ($examFields as $f) { $examRaw += floatval($mark->$f ?? 0); }
   523→                $row['ca_total'] = $caRawTotal > 0 ? round(($caRaw / $caRawTotal) * $caWeight, $precision) : 0;
   524→                $row['exam_total'] = min($examRaw, $examWeight);
   525→                $row['grand_total'] = round($row['ca_total'] + $row['exam_total'], $precision);
   526→            } else {
   527→                $row['ca_total'] = 0;
   528→                $row['exam_total'] = 0;
   529→                $row['grand_total'] = 0;
   530→            }
   531→            $rows[]=$row;
   532→        }
   533→        $subject=Subject::find($subjectId); $term=Term::find($termId); $class=ClassRoom::find($classId); $section=Section::find($sectionId);
   534→        return response()->json(['students'=>$rows,'markFields'=>$markFields,
   535→            'subject'=>$subject?$subject->name:'','term'=>$term?$term->name:'',
   536→            'class'=>$class?$class->name:'','section'=>$section?$section->name:'']);
   537→    }

   539→    public function apiSave(Request $request) {
   540→        // Relaxed validation — auto-save sends minimal data
   541→        $request->validate([
   542→            'student_id' => 'required',
   543→            'subject_id' => 'required',
   544→            'term_id' => 'required',
   545→        ]);
   546→
   547→        // Touch session to keep it alive during mark entry
   548→        $request->session()->put('_last_mark_save', time());
   549→        // Force-save session to database immediately (critical for database driver)
   550→        $request->session()->save();
   551→
   552→        $studentId = $request->input('student_id');
   553→        $subjectId = $request->input('subject_id');
   554→        $ayId = $request->input('academic_year_id') ?: null;
   555→        $termId = $request->input('term_id');
   556→
   557→        // ── Authorization check for teachers ──
   558→        $teacher = $this->getTeacherForUser();
   559→        if ($teacher) {
   560→            $classId = $request->input('class_id');
   561→            $sectionId = $request->input('section_id');
   562→
   563→            // Resolve class_id from class_grade if needed
   564→            if (empty($classId) && $request->filled('class_grade')) {
   565→                $classRoom = ClassRoom::where('name', $request->input('class_grade'))->first();
   566→                if ($classRoom) $classId = $classRoom->id;
   567→            }
   568→            // Resolve section_id from section if needed
   569→            if (empty($sectionId) && $request->filled('section') && !empty($classId)) {
   570→                $sectionModel = Section::where('class_id', $classId)
   571→                    ->where('name', $request->input('section'))->first();
   572→                if ($sectionModel) $sectionId = $sectionModel->id;
   573→            }
   574→
   575→            // Check if teacher is assigned to this class+section+subject
   576→            $isAssigned = TeacherAssignment::where('teacher_id', $teacher->id)
   577→                ->where('class_id', $classId)
   578→                ->where('subject_id', $subjectId)
   579→                ->where(function($q) use ($sectionId) {
   580→                    $q->where('section_id', $sectionId)->orWhereNull('section_id');
   581→                })
   582→                ->exists();
   583→
   584→            // Or check if teacher is homeroom for this class or section
   585→            $isHomeroomClass = !empty($classId) && $teacher->classRooms()->where('id', $classId)->exists();
   586→            $isHomeroomSection = !empty($sectionId) && $teacher->sections()->where('id', $sectionId)->exists();
   587→
   588→            if (!$isAssigned && !$isHomeroomClass && !$isHomeroomSection) {
   589→                return response()->json([
   590→                    'success' => false,
   591→                    'error' => 'You are not authorized to enter marks for this class/section/subject.',
   592→                ], 403);
   593→            }
   594→
   595→            // ── Disallowal check: branch principal can disallow specific teachers ──
   596→            $effectiveAyId = $ayId;
   597→            $effectiveTermId = $termId;
   598→            if (!$effectiveAyId) {
   599→                $activeAy = AcademicYear::where('is_current', true)->first();
   600→                $effectiveAyId = $activeAy?->id;
   601→            }
   602→            if (!$effectiveTermId && $effectiveAyId) {
   603→                $activeTerm = Term::where('academic_year_id', $effectiveAyId)->where('is_active', true)->first();
   604→                $effectiveTermId = $activeTerm?->id;
   605→            }
   606→
   607→            $isDisallowed = MarkEntryDisallowal::isDisallowed(
   608→                $teacher->id,
   609→                $classId,
   610→                $sectionId,
   611→                $subjectId,
   612→                $effectiveAyId,
   613→                $effectiveTermId
   614→            );
   615→
   616→            if ($isDisallowed) {
   617→                return response()->json([
   618→                    'success' => false,
   619→                    'error' => 'Your mark entry permission has been restricted for this class/section/subject. Contact your branch principal for assistance.',
   620→                    'is_disallowed' => true,
   621→                ], 403);
   622→            }
   623→        }
   624→
   625→        // ── Mark Entry Lock & Permission Check ──
   626→        $user = auth()->user();
   627→        $isAdmin = $user && in_array($user->role, ['admin', 'super_admin']);
   628→
   629→        if (!$isAdmin) {
   630→            // Resolve class_id for lock check
   631→            $lockClassId = $request->input('class_id');
   632→            if (empty($lockClassId) && $request->filled('class_grade')) {
   633→                $cr = ClassRoom::where('name', $request->input('class_grade'))->first();
   634→                if ($cr) $lockClassId = $cr->id;
   635→            }
   636→            // Get branch_id from the class
   637→            $branchId = null;
   638→            if ($lockClassId) {
   639→                $classModel = ClassRoom::find($lockClassId);
   640→                if ($classModel) $branchId = $classModel->branch_id;
   641→            }
   642→            // Fallback: use user's branch
   643→            if (!$branchId && $user) {
   644→                $branchId = $user->branch_id;
   645→            }
   646→
   647→            $isLocked = $branchId && $ayId && $termId && MarkEntryLock::isLocked($branchId, $ayId, $termId);
   648→
   649→            if ($isLocked) {
   650→                // Mark entry is locked — check if teacher has special permission
   651→                $hasPermission = false;
   652→                if ($teacher) {
   653→                    $hasPermission = MarkEntryPermission::hasPermission(
   654→                        $teacher->id, $studentId, $subjectId, $ayId, $termId
   655→                    );
   656→                }
   657→
   658→                if (!$hasPermission) {
   659→                    return response()->json([
   660→                        'success' => false,
   661→                        'error' => 'Mark entry is locked for this term. You do not have permission to edit these marks.',
   662→                        'is_locked' => true,
   663→                    ], 403);
   664→                }
   665→            }
   666→        }
   667→
   668→        // Collect only the mark fields that were actually sent (from config)
   669→        $markFieldNames = array_map(function($f) { return $f['col']; }, MarkEntry::getMarkFields());
   670→        $data = [];
   671→        $data['student_id'] = $studentId;
   672→        $data['subject_id'] = $subjectId;
   673→        $data['academic_year_id'] = $ayId;
   674→        $data['term_id'] = $termId;
   675→
   676→        // Copy explicit fields from request
   677→        foreach (['class_id','section_id','class_grade','section','exam_id','grade','remarks'] as $f) {
   678→            if ($request->has($f)) {
   679→                $data[$f] = $request->input($f) ?: null;
   680→            }
   681→        }
   682→
   683→        // Resolve class_id and section_id from class_grade/section if not provided
   684→        if (empty($data['class_id']) && !empty($data['class_grade'])) {
   685→            $classRoom = ClassRoom::where('name', $data['class_grade'])->first();
   686→            if ($classRoom) {
   687→                $data['class_id'] = $classRoom->id;
   688→            }
   689→        }
   690→        if (empty($data['section_id']) && !empty($data['section']) && !empty($data['class_id'])) {
   691→            $sectionModel = Section::where('class_id', $data['class_id'])
   692→                ->where('name', $data['section'])->first();
   693→            if ($sectionModel) {
   694→                $data['section_id'] = $sectionModel->id;
   695→            }
   696→        }
   697→
   698→        // Set teacher_id — use user_id FK first, then email fallback
   699→        $teacherId = null;
   700→        if (auth()->check()) {
   701→            $t = Teacher::where('user_id', auth()->user()->id)->first();
   702→            if (!$t) {
   703→                $t = Teacher::where('email', auth()->user()->email)->first();
   704→            }
   705→            if ($t) {
   706→                $teacherId = $t->id;
   707→            }
   708→        }
   709→        $data['teacher_id'] = $teacherId;
   710→
   711→        // ── Load existing record FIRST so calcTotals has complete data ──
   712→        $existingQuery = MarkEntry::where('student_id', $studentId)
   713→            ->where('subject_id', $subjectId)
   714→            ->where('term_id', $termId);
   715→        if (!empty($ayId)) {
   716→            $existingQuery->where('academic_year_id', $ayId);
   717→        } else {
   718→            $existingQuery->whereNull('academic_year_id');
   719→        }
   720→        $existing = $existingQuery->first();
   721→
   722→        // Seed $data with existing values so calcTotals works with ALL fields
   723→        if ($existing) {
   724→            foreach ($markFieldNames as $f) {
   725→                if (!isset($data[$f])) {
   726→                    $data[$f] = $existing->$f;
   727→                }
   728→            }
   729→        }
   730→
   731→        // Handle single-field auto-save (mark_key + mark_value) — override the one field
   732→        if ($request->filled('mark_key') && $request->has('mark_value')) {
   733→            $markKey = $request->mark_key;
   734→            $markValue = $request->mark_value;
   735→            $data[$markKey] = ($markValue === '' || $markValue === null) ? null : $markValue;
   736→        } else {
   737→            // Full save — copy all mark fields from request
   738→            foreach ($markFieldNames as $f) {
   739→                if ($request->has($f)) {
   740→                    $val = $request->input($f);
   741→                    $data[$f] = ($val === '' || $val === null) ? null : $val;
   742→                }
   743→            }
   744→        }
   745→
   746→        // Calculate totals — now $data has ALL mark fields, not just the changed one
   747→        $data = MarkEntry::calcTotals($data);
   748→
   749→        // Set marks_obtained = grand_total for backward compatibility (column is NOT NULL in original migration)
   750→        $data['marks_obtained'] = $data['grand_total'] ?? 0;
   751→
   752→        try {
   753→            // $existing was already loaded above (before calcTotals)
   754→            if ($existing) {
   755→                // Only update the fields we actually have data for
   756→                $updateData = array_filter($data, function($value, $key) use ($markFieldNames) {
   757→                    // Always update these core fields
   758→                    if (in_array($key, ['student_id','subject_id','academic_year_id','term_id',
   759→                        'class_id','section_id','teacher_id','marks_obtained',
   760→                        'ca_total','exam_total','grand_total'])) return true;
   761→                    // Update mark fields only if they were explicitly set
   762→                    if (in_array($key, $markFieldNames)) return true;
   763→                    // Update other fields like class_grade, section, etc.
   764→                    if (in_array($key, ['class_grade','section','exam_id','grade','remarks'])) return true;
   765→                    return false;
   766→                }, ARRAY_FILTER_USE_BOTH);
   767→
   768→                $existing->update($updateData);
   769→                $entry = $existing->fresh();
   770→            } else {
   771→                // Set defaults for required fields on create
   772→                if (!isset($data['marks_obtained'])) $data['marks_obtained'] = 0;
   773→                $entry = MarkEntry::create($data);
   774→            }
   775→        } catch (\Illuminate\Database\QueryException $e) {
   776→            Log::error('MarkEntry save error: ' . $e->getMessage(), [
   777→                'data' => $data,
   778→                'trace' => $e->getTraceAsString()
   779→            ]);
   780→            return response()->json([
   781→                'success' => false,
   782→                'error' => 'Database error: ' . $e->getMessage(),
   783→            ], 500);
   784→        } catch (\Throwable $e) {
   785→            Log::error('MarkEntry save exception: ' . $e->getMessage(), [
   786→                'class' => get_class($e),
   787→                'data' => $data
   788→            ]);
   789→            return response()->json([
   790→                'success' => false,
   791→                'error' => 'Error: ' . $e->getMessage(),
   792→            ], 500);
   793→        }
   794→
   795→        return response()->json([
   796→            'success' => true,
   797→            'entry' => $entry,
   798→            'ca_total' => $entry->ca_total,
   799→            'exam_total' => $entry->exam_total,
   800→            'grand_total' => $entry->grand_total,
   801→            'grade' => $entry->grade ?? null,
   802→            'csrf_token' => csrf_token(),
   803→        ]);
   804→    }

   806→    /**
   807→     * Bulk save marks for multiple students at once.
   808→     * The teacher selects a mark field (e.g., 'ca1', 'exam1') and enters
   809→     * values for all students in a table. This endpoint saves them all.
   810→     */
   811→    public function apiBulkSave(Request $request)
   812→    {
   813→        // Handle JSON body — Laravel doesn't auto-parse JSON arrays into $request->all()
   814→        $input = $request->all();
   815→        
   816→        // If marks was sent as a JSON string, decode it
   817→        if (is_string($input['marks'] ?? null)) {
   818→            $input['marks'] = json_decode($input['marks'], true);
   819→        }
   820→
   821→        // Manual validation — return JSON errors, not redirects
   822→        $errors = [];
   823→        if (empty($input['subject_id'])) $errors['subject_id'] = ['Subject is required.'];
   824→        if (empty($input['term_id'])) $errors['term_id'] = ['Term is required.'];
   825→        if (empty($input['mark_key'])) $errors['mark_key'] = ['Mark field is required.'];
   826→        if (empty($input['marks']) || !is_array($input['marks'])) $errors['marks'] = ['Marks array is required.'];
   827→
   828→        if (!empty($errors)) {
   829→            return response()->json([
   830→                'success' => false,
   831→                'error' => 'Validation failed: ' . json_encode($errors),
   832→                'errors' => $errors,
   833→            ], 422);
   834→        }
   835→
   836→        // Touch session to keep it alive
   837→        $request->session()->put('_last_mark_save', time());
   838→        $request->session()->save();
   839→
   840→        $subjectId = $input['subject_id'];
   841→        $ayId = $input['academic_year_id'] ?? null;
   842→        $termId = $input['term_id'];
   843→        $markKey = $input['mark_key'];
   844→        $classId = $input['class_id'] ?? null;
   845→        $sectionId = $input['section_id'] ?? null;
   846→        $marks = $input['marks'];
   847→
   848→        // Authorization check for teachers
   849→        $teacher = $this->getTeacherForUser();
   850→        if ($teacher) {
   851→            $isAssigned = TeacherAssignment::where('teacher_id', $teacher->id)
   852→                ->where('class_id', $classId)
   853→                ->where('subject_id', $subjectId)
   854→                ->where(function($q) use ($sectionId) {
   855→                    $q->where('section_id', $sectionId)->orWhereNull('section_id');
   856→                })
   857→                ->exists();
   858→
   859→            $isHomeroomClass = $teacher->classRooms()->where('id', $classId)->exists();
   860→            $isHomeroomSection = $sectionId && $teacher->sections()->where('id', $sectionId)->exists();
   861→
   862→            if (!$isAssigned && !$isHomeroomClass && !$isHomeroomSection) {
   863→                return response()->json(['success' => false, 'error' => 'Not authorized.'], 403);
   864→            }
   865→        }
   866→
   867→        $saved = 0;
   868→        $errors = [];
   869→
   870→        foreach ($marks as $item) {
   871→            $studentId = $item['student_id'];
   872→            $value = $item['value'];
   873→            if ($value === '' || $value === null) {
   874→                $value = null;
   875→            }
   876→
   877→            // Load existing record
   878→            $existingQuery = MarkEntry::where('student_id', $studentId)
   879→                ->where('subject_id', $subjectId)
   880→                ->where('term_id', $termId);
   881→            if ($ayId) {
   882→                $existingQuery->where('academic_year_id', $ayId);
   883→            } else {
   884→                $existingQuery->whereNull('academic_year_id');
   885→            }
   886→            $existing = $existingQuery->first();
   887→
   888→            // Build data with ALL fields (seed from existing so calcTotals works)
   889→            $data = [
   890→                'student_id' => $studentId,
   891→                'subject_id' => $subjectId,
   892→                'academic_year_id' => $ayId,
   893→                'term_id' => $termId,
   894→                'class_id' => $classId,
   895→                'section_id' => $sectionId,
   896→                'teacher_id' => $teacher?->id,
   897→            ];
   898→
   899→            if ($existing) {
   900→                $markFieldNames = array_map(fn($f) => $f['col'], MarkEntry::getMarkFields());
   901→                foreach ($markFieldNames as $f) {
   902→                    $data[$f] = $existing->$f;
   903→                }
   904→            }
   905→
   906→            // Override the single field being bulk-edited
   907→            $data[$markKey] = $value;
   908→
   909→            // Calculate totals
   910→            $data = MarkEntry::calcTotals($data);
   911→            $data['marks_obtained'] = $data['grand_total'] ?? 0;
   912→
   913→            try {
   914→                if ($existing) {
   915→                    $existing->update($data);
   916→                } else {
   917→                    MarkEntry::create($data);
   918→                }
   919→                $saved++;
   920→            } catch (\Throwable $e) {
   921→                $errors[] = "Student {$studentId}: " . $e->getMessage();
   922→            }
   923→        }
   924→
   925→        return response()->json([
   926→            'success' => true,
   927→            'saved' => $saved,
   928→            'errors' => $errors,
   929→            'csrf_token' => csrf_token(),
   930→        ]);
   931→    }
