    25→     * Tries user_id FK first, then falls back to email match.

    28→    private function getTeacherForUser()
    29→    {
    30→        $user = auth()->user();
    31→        if (!$user || $user->role !== 'teacher') return null;
    32→
    33→        // Try user_id FK first
    34→        $teacher = Teacher::where('user_id', $user->id)->first();
    35→        if (!$teacher) {
    36→            // Fall back to email match (legacy)
    37→            $teacher = Teacher::where('email', $user->email)->first();
    38→        }
    39→        return $teacher;

   501→            $mark = $existingMarks->get($s->student_id);
   502→            $row = ['student_id'=>$s->student_id,'student_name'=>$s->student_name??'Unknown','roll_number'=>$s->roll_number,'gender'=>$s->gender];
   503→            foreach ($markFields as $field) { $col=$field['col']; $row[$col]=$mark?floatval($mark->$col):null; }
   504→            // Recalculate totals from raw fields (don't trust stored values)
   505→            if ($mark) {
   506→                $caRaw = 0;
   507→                foreach ($caFields as $f) { $caRaw += floatval($mark->$f ?? 0); }
   508→                $examRaw = 0;
   509→                foreach ($examFields as $f) { $examRaw += floatval($mark->$f ?? 0); }
   510→                $row['ca_total'] = $caRawTotal > 0 ? round(($caRaw / $caRawTotal) * $caWeight, $precision) : 0;
   511→                $row['exam_total'] = min($examRaw, $examWeight);
   512→                $row['grand_total'] = round($row['ca_total'] + $row['exam_total'], $precision);
   513→            } else {
   514→                $row['ca_total'] = 0;
   515→                $row['exam_total'] = 0;
   516→                $row['grand_total'] = 0;
   517→            }
   518→            $rows[]=$row;
   519→        }
   520→        $subject=Subject::find($subjectId); $term=Term::find($termId); $class=ClassRoom::find($classId); $section=Section::find($sectionId);
   521→        return response()->json(['students'=>$rows,'markFields'=>$markFields,
   522→            'subject'=>$subject?$subject->name:'','term'=>$term?$term->name:'',
   523→            'class'=>$class?$class->name:'','section'=>$section?$section->name:'']);
   524→    }

   526→    public function apiSave(Request $request) {
   527→        // Relaxed validation — auto-save sends minimal data
   528→        $request->validate([
   529→            'student_id' => 'required',
   530→            'subject_id' => 'required',
   531→            'term_id' => 'required',
   532→        ]);
   533→
   534→        // Touch session to keep it alive during mark entry
   535→        $request->session()->put('_last_mark_save', time());
   536→        // Force-save session to database immediately (critical for database driver)
   537→        $request->session()->save();
   538→
   539→        $studentId = $request->input('student_id');
   540→        $subjectId = $request->input('subject_id');
   541→        $ayId = $request->input('academic_year_id') ?: null;
   542→        $termId = $request->input('term_id');
   543→
   544→        // ── Authorization check for teachers ──
   545→        $teacher = $this->getTeacherForUser();
   546→        if ($teacher) {
   547→            $classId = $request->input('class_id');
   548→            $sectionId = $request->input('section_id');
   549→
   550→            // Resolve class_id from class_grade if needed
   551→            if (empty($classId) && $request->filled('class_grade')) {
   552→                $classRoom = ClassRoom::where('name', $request->input('class_grade'))->first();
   553→                if ($classRoom) $classId = $classRoom->id;
   554→            }
   555→            // Resolve section_id from section if needed
   556→            if (empty($sectionId) && $request->filled('section') && !empty($classId)) {
   557→                $sectionModel = Section::where('class_id', $classId)
   558→                    ->where('name', $request->input('section'))->first();
   559→                if ($sectionModel) $sectionId = $sectionModel->id;
   560→            }
   561→
   562→            // Check if teacher is assigned to this class+section+subject
   563→            $isAssigned = TeacherAssignment::where('teacher_id', $teacher->id)
   564→                ->where('class_id', $classId)
   565→                ->where('subject_id', $subjectId)
   566→                ->where(function($q) use ($sectionId) {
   567→                    $q->where('section_id', $sectionId)->orWhereNull('section_id');
   568→                })
   569→                ->exists();
   570→
   571→            // Or check if teacher is homeroom for this class or section
   572→            $isHomeroomClass = !empty($classId) && $teacher->classRooms()->where('id', $classId)->exists();
   573→            $isHomeroomSection = !empty($sectionId) && $teacher->sections()->where('id', $sectionId)->exists();
   574→
   575→            if (!$isAssigned && !$isHomeroomClass && !$isHomeroomSection) {
   576→                return response()->json([
   577→                    'success' => false,
   578→                    'error' => 'You are not authorized to enter marks for this class/section/subject.',
   579→                ], 403);
   580→            }
   581→
   582→            // ── Disallowal check: branch principal can disallow specific teachers ──
   583→            $effectiveAyId = $ayId;
   584→            $effectiveTermId = $termId;
   585→            if (!$effectiveAyId) {
   586→                $activeAy = AcademicYear::where('is_current', true)->first();
   587→                $effectiveAyId = $activeAy?->id;
   588→            }
   589→            if (!$effectiveTermId && $effectiveAyId) {
   590→                $activeTerm = Term::where('academic_year_id', $effectiveAyId)->where('is_active', true)->first();
   591→                $effectiveTermId = $activeTerm?->id;
   592→            }
   593→
   594→            $isDisallowed = MarkEntryDisallowal::isDisallowed(
   595→                $teacher->id,
   596→                $classId,
   597→                $sectionId,
   598→                $subjectId,
   599→                $effectiveAyId,
   600→                $effectiveTermId
   601→            );
   602→
   603→            if ($isDisallowed) {
   604→                return response()->json([
   605→                    'success' => false,
   606→                    'error' => 'Your mark entry permission has been restricted for this class/section/subject. Contact your branch principal for assistance.',
   607→                    'is_disallowed' => true,
   608→                ], 403);
   609→            }
   610→        }
   611→
   612→        // ── Mark Entry Lock & Permission Check ──
   613→        $user = auth()->user();
   614→        $isAdmin = $user && in_array($user->role, ['admin', 'super_admin']);
   615→
   616→        if (!$isAdmin) {
   617→            // Resolve class_id for lock check
   618→            $lockClassId = $request->input('class_id');
   619→            if (empty($lockClassId) && $request->filled('class_grade')) {
   620→                $cr = ClassRoom::where('name', $request->input('class_grade'))->first();
   621→                if ($cr) $lockClassId = $cr->id;
   622→            }
   623→            // Get branch_id from the class
   624→            $branchId = null;
   625→            if ($lockClassId) {
   626→                $classModel = ClassRoom::find($lockClassId);
   627→                if ($classModel) $branchId = $classModel->branch_id;
   628→            }
   629→            // Fallback: use user's branch
   630→            if (!$branchId && $user) {
   631→                $branchId = $user->branch_id;
   632→            }
   633→
   634→            $isLocked = $branchId && $ayId && $termId && MarkEntryLock::isLocked($branchId, $ayId, $termId);
   635→
   636→            if ($isLocked) {
   637→                // Mark entry is locked — check if teacher has special permission
   638→                $hasPermission = false;
   639→                if ($teacher) {
   640→                    $hasPermission = MarkEntryPermission::hasPermission(
   641→                        $teacher->id, $studentId, $subjectId, $ayId, $termId
   642→                    );
   643→                }
   644→
   645→                if (!$hasPermission) {
   646→                    return response()->json([
   647→                        'success' => false,
   648→                        'error' => 'Mark entry is locked for this term. You do not have permission to edit these marks.',
   649→                        'is_locked' => true,
   650→                    ], 403);
   651→                }
   652→            }
   653→        }
   654→
   655→        // Collect only the mark fields that were actually sent (from config)
   656→        $markFieldNames = array_map(function($f) { return $f['col']; }, MarkEntry::getMarkFields());
   657→        $data = [];
   658→        $data['student_id'] = $studentId;
   659→        $data['subject_id'] = $subjectId;
   660→        $data['academic_year_id'] = $ayId;
   661→        $data['term_id'] = $termId;
   662→
   663→        // Copy explicit fields from request
   664→        foreach (['class_id','section_id','class_grade','section','exam_id','grade','remarks'] as $f) {
   665→            if ($request->has($f)) {
   666→                $data[$f] = $request->input($f) ?: null;
   667→            }
   668→        }
   669→
   670→        // Resolve class_id and section_id from class_grade/section if not provided
   671→        if (empty($data['class_id']) && !empty($data['class_grade'])) {
   672→            $classRoom = ClassRoom::where('name', $data['class_grade'])->first();
   673→            if ($classRoom) {
   674→                $data['class_id'] = $classRoom->id;
   675→            }
   676→        }
   677→        if (empty($data['section_id']) && !empty($data['section']) && !empty($data['class_id'])) {
   678→            $sectionModel = Section::where('class_id', $data['class_id'])
   679→                ->where('name', $data['section'])->first();
   680→            if ($sectionModel) {
   681→                $data['section_id'] = $sectionModel->id;
   682→            }
   683→        }
   684→
   685→        // Set teacher_id — use user_id FK first, then email fallback
   686→        $teacherId = null;
   687→        if (auth()->check()) {
   688→            $t = Teacher::where('user_id', auth()->user()->id)->first();
   689→            if (!$t) {
   690→                $t = Teacher::where('email', auth()->user()->email)->first();
   691→            }
   692→            if ($t) {
   693→                $teacherId = $t->id;
   694→            }
   695→        }
   696→        $data['teacher_id'] = $teacherId;
   697→
   698→        // ── Load existing record FIRST so calcTotals has complete data ──
   699→        $existingQuery = MarkEntry::where('student_id', $studentId)
   700→            ->where('subject_id', $subjectId)
   701→            ->where('term_id', $termId);
   702→        if (!empty($ayId)) {
   703→            $existingQuery->where('academic_year_id', $ayId);
   704→        } else {
   705→            $existingQuery->whereNull('academic_year_id');
   706→        }
   707→        $existing = $existingQuery->first();
   708→
   709→        // Seed $data with existing values so calcTotals works with ALL fields
   710→        if ($existing) {
   711→            foreach ($markFieldNames as $f) {
   712→                if (!isset($data[$f])) {
   713→                    $data[$f] = $existing->$f;
   714→                }
   715→            }
   716→        }
   717→
   718→        // Handle single-field auto-save (mark_key + mark_value) — override the one field
   719→        if ($request->filled('mark_key') && $request->has('mark_value')) {
   720→            $markKey = $request->mark_key;
   721→            $markValue = $request->mark_value;
   722→            $data[$markKey] = ($markValue === '' || $markValue === null) ? null : $markValue;
   723→        } else {
   724→            // Full save — copy all mark fields from request
   725→            foreach ($markFieldNames as $f) {
   726→                if ($request->has($f)) {
   727→                    $val = $request->input($f);
   728→                    $data[$f] = ($val === '' || $val === null) ? null : $val;
   729→                }
   730→            }
   731→        }
   732→
   733→        // Calculate totals — now $data has ALL mark fields, not just the changed one
   734→        $data = MarkEntry::calcTotals($data);
   735→
   736→        // Set marks_obtained = grand_total for backward compatibility (column is NOT NULL in original migration)
   737→        $data['marks_obtained'] = $data['grand_total'] ?? 0;
   738→
   739→        try {
   740→            // $existing was already loaded above (before calcTotals)
   741→            if ($existing) {
   742→                // Only update the fields we actually have data for
   743→                $updateData = array_filter($data, function($value, $key) use ($markFieldNames) {
   744→                    // Always update these core fields
   745→                    if (in_array($key, ['student_id','subject_id','academic_year_id','term_id',
   746→                        'class_id','section_id','teacher_id','marks_obtained',
   747→                        'ca_total','exam_total','grand_total'])) return true;
   748→                    // Update mark fields only if they were explicitly set
   749→                    if (in_array($key, $markFieldNames)) return true;
   750→                    // Update other fields like class_grade, section, etc.
   751→                    if (in_array($key, ['class_grade','section','exam_id','grade','remarks'])) return true;
   752→                    return false;
   753→                }, ARRAY_FILTER_USE_BOTH);
   754→
   755→                $existing->update($updateData);
   756→                $entry = $existing->fresh();
   757→            } else {
   758→                // Set defaults for required fields on create
   759→                if (!isset($data['marks_obtained'])) $data['marks_obtained'] = 0;
   760→                $entry = MarkEntry::create($data);
   761→            }
   762→        } catch (\Illuminate\Database\QueryException $e) {
   763→            Log::error('MarkEntry save error: ' . $e->getMessage(), [
   764→                'data' => $data,
   765→                'trace' => $e->getTraceAsString()
   766→            ]);
   767→            return response()->json([
   768→                'success' => false,
   769→                'error' => 'Database error: ' . $e->getMessage(),
   770→            ], 500);
   771→        } catch (\Throwable $e) {
   772→            Log::error('MarkEntry save exception: ' . $e->getMessage(), [
   773→                'class' => get_class($e),
   774→                'data' => $data
   775→            ]);
   776→            return response()->json([
   777→                'success' => false,
   778→                'error' => 'Error: ' . $e->getMessage(),
   779→            ], 500);
   780→        }
   781→
   782→        return response()->json([
   783→            'success' => true,
   784→            'entry' => $entry,
   785→            'ca_total' => $entry->ca_total,
   786→            'exam_total' => $entry->exam_total,
   787→            'grand_total' => $entry->grand_total,
   788→            'grade' => $entry->grade ?? null,
   789→            'csrf_token' => csrf_token(),  // Fresh token — prevents 419 on next save
   790→        ]);
   791→    }
