   332→    function toggleQuestionType() {
   333→        var type = $('#questionType').val();
   334→        if (type === 'multiple_choice') {
   335→            $('#optionsCard').show();

   345→            $('#tfCard').show();
   346→            $('input[name="correct_tf"]').prop('disabled', false);
   347→        } else {

   349→            $('#optionsCard').hide();
   350→            $('#optionsContainer input').prop('disabled', true);

   352→            $('input[name="correct_tf"]').prop('checked', false).prop('disabled', true);
   353→        }
