   296→    // a keepalive ping could invalidate the CSRF token of an in-flight
   297→    // mark-entry save request, leading to 419 errors.
   298→    // The CSRF token is refreshed only when:
   299→    //   1. A save response returns a fresh token (mark entry auto-save)
   300→    //   2. A 419 error triggers a token refresh + retry
   301→    Route::match(['get', 'post'], '/keepalive', function () {
   302→        $request = request();
   303→
   304→        // Touch the session to keep it alive
   305→        $request->session()->put('_last_keepalive', time());
   306→
   307→        // Force-save session to database immediately
   308→        $request->session()->save();
   309→
   310→        return response()->json([
   311→            'csrf_token' => csrf_token(),
   312→            'timestamp' => time(),
   313→            'driver' => config('session.driver'),
   314→            'lifetime' => config('session.lifetime'),
   315→            'ok' => true,
   316→        ])->header('Cache-Control', 'no-store, no-cache, must-revalidate')
   317→          ->header('Pragma', 'no-cache')
   318→          ->header('X-Session-Refreshed', '1');
   319→    })->name('keepalive');

   321→    // Session Diagnostic — helps verify session is working correctly
   322→    Route::get('/session-diagnostic', function () {
   323→        $request = request();
   324→        $sessionId = $request->session()->getId();
   325→        $handler = $request->session()->getHandler();
   326→        $handlerClass = get_class($handler);
   327→
   328→        $diagnostics = [
   329→            'timestamp' => date('Y-m-d H:i:s'),
   330→            'php_version' => PHP_VERSION,
   331→            'session_id' => $sessionId,
   332→            'session_driver_config' => config('session.driver'),
   333→            'session_lifetime_config' => config('session.lifetime'),
   334→            'session_cookie_config' => config('session.cookie'),
   335→            'session_encrypt_config' => config('session.encrypt'),
   336→            'session_lottery_config' => config('session.lottery'),
   337→            'handler_class' => $handlerClass,
   338→            'php_gc_maxlifetime' => ini_get('session.gc_maxlifetime'),
   339→            'php_gc_probability' => ini_get('session.gc_probability'),
   340→            'php_gc_divisor' => ini_get('session.gc_divisor'),
   341→            'php_session_cookie_lifetime' => ini_get('session.cookie_lifetime'),
   342→            'session_data_keys' => array_keys($request->session()->all()),
   343→            'auth_user_id' => auth()->id(),
   344→            'auth_user_role' => auth()->user()?->role,
   345→        ];
   346→
   347→        // Database driver specific diagnostics
   348→        if (config('session.driver') === 'database') {
   349→            try {
   350→                $dbSession = \DB::table('sessions')->where('id', $sessionId)->first();
   351→                $diagnostics['db_session_found'] = $dbSession ? true : false;
   352→                $diagnostics['db_session_last_activity'] = $dbSession ? $dbSession->last_activity : null;
   353→                $diagnostics['db_session_last_activity_readable'] = $dbSession ? date('Y-m-d H:i:s', $dbSession->last_activity) : null;
   354→                $diagnostics['db_session_user_id'] = $dbSession ? $dbSession->user_id : null;
   355→                $diagnostics['db_session_ip'] = $dbSession ? $dbSession->ip_address : null;
   356→                $diagnostics['db_total_sessions'] = \DB::table('sessions')->count();
   357→                $diagnostics['db_session_age_seconds'] = $dbSession ? (time() - $dbSession->last_activity) : null;
   358→                $diagnostics['db_session_expires_in_seconds'] = $dbSession ? (config('session.lifetime') * 60 - (time() - $dbSession->last_activity)) : null;
   359→            } catch (\Throwable $e) {
   360→                $diagnostics['db_error'] = $e->getMessage();
   361→            }
   362→        }
   363→
   364→        return response()->json($diagnostics, 200, [], JSON_PRETTY_PRINT);
   365→    })->name('session-diagnostic');
