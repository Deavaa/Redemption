   240→// ── Static file fallback for subdirectory hosting ──
   241→// When running without /public in the URL (e.g., http://localhost/redemption/css/admin.css),
   242→// the .htaccess routes ALL requests through index.php. Laravel needs to serve static
   243→// files (CSS, JS, images, fonts, etc.) from the public/ directory.
   244→// This route catches requests for common static file extensions and serves them.
   245→Route::get('/{path}', function ($path = null) {
   246→    // Fallback route: if no path, redirect to login
   247→    if (empty($path)) {
   248→        return redirect()->route('login');
   249→    }
   250→
   251→    $publicPath = public_path($path);
   252→
   253→    // Only serve if the file actually exists in public/
   254→    if (file_exists($publicPath) && is_file($publicPath)) {
   255→        $mimeTypes = [
   256→            'css' => 'text/css',
   257→            'js' => 'application/javascript',
   258→            'png' => 'image/png',
   259→            'jpg' => 'image/jpeg',
   260→            'jpeg' => 'image/jpeg',
   261→            'gif' => 'image/gif',
   262→            'svg' => 'image/svg+xml',
   263→            'ico' => 'image/x-icon',
   264→            'webp' => 'image/webp',
   265→            'woff' => 'font/woff',
   266→            'woff2' => 'font/woff2',
   267→            'ttf' => 'font/ttf',
   268→            'eot' => 'application/vnd.ms-fontobject',
   269→            'otf' => 'font/otf',
   270→            'json' => 'application/json',
   271→            'xml' => 'application/xml',
   272→            'txt' => 'text/plain',
   273→            'pdf' => 'application/pdf',
   274→            'apk' => 'application/vnd.android.package-archive',
   275→            'idsig' => 'application/octet-stream',
   276→            'webmanifest' => 'application/manifest+json',
   277→            'map' => 'application/json',
   278→        ];
   279→
   280→        $ext = strtolower(pathinfo($publicPath, PATHINFO_EXTENSION));
   281→        $mime = $mimeTypes[$ext] ?? 'application/octet-stream';
   282→
   283→        return response()->file($publicPath, [
   284→            'Content-Type' => $mime,
   285→            'Cache-Control' => 'public, max-age=86400',
   286→        ]);
   287→    }
   288→
   289→    abort(404);
   290→})->where('path', '.*')->fallback();

   292→Route::prefix('admin')->name('admin.')->middleware(['auth', 'admin', 'branch-scope'])->group(function () {
   293→    // Session Keepalive — touches session to keep it alive
   294→    // NOTE: We do NOT regenerate the CSRF token here anymore.
   295→    // Regenerating the token on every keepalive caused a race condition:
   296→    // a keepalive ping could invalidate the CSRF token of an in-flight
   297→    // mark-entry save request, leading to 419 errors.
   298→    // The CSRF token is refreshed only when:
   299→    //   1. A save response returns a fresh token (mark entry auto-save)
   300→    //   2. A 419 error triggers a token refresh + retry
   301→    Route::match(['get', 'post'], '/keepalive', function () {
   302→        $request = request();
   303→
   304→        // Touch the session to keep it alive
   305→        $request->session()->put('_last_keepalive', time());
   306→
   307→        // Force-save session to database immediately
   308→        $request->session()->save();
   309→
   310→        return response()->json([
   311→            'csrf_token' => csrf_token(),
   312→            'timestamp' => time(),
   313→            'driver' => config('session.driver'),
   314→            'lifetime' => config('session.lifetime'),
   315→            'ok' => true,
   316→        ])->header('Cache-Control', 'no-store, no-cache, must-revalidate')
   317→          ->header('Pragma', 'no-cache')
   318→          ->header('X-Session-Refreshed', '1');
   319→    })->name('keepalive');
